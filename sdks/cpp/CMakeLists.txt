# The C++ SDK of design §44 §9 row 12 (SDK2 Task 11, issue #295).
#
# C++17, a static library called `loams`, and ctest. **No gRPC.** The wire is
# Connect over HTTP (design §44 §4, D600): one port serves Connect, gRPC and
# gRPC-Web, and a C++ client that needs a full gRPC stack to make an HTTP POST is
# a heavier install for the same bytes. The four content types the SDK speaks are
# Connect JSON and proto, and gRPC-Web proto and JSON — all of them HTTP/1.1.
#
# Two dependencies, both preinstalled on the machines and the CI runners that
# develop this repository, and both licence-compatible (Apache-2.0 / MIT / BSD):
# protobuf (BSD-3-Clause) and libcurl (curl-licence, MIT-like). `DEPENDENCIES.md`
# records why each is needed; protobuf is the message codec and the generated
# stubs' own runtime, and libcurl is the HTTP layer behind `HttpTransport`.
#
# The generated stubs under `gen/` are **committed** and compiled from here, so a
# consumer needs no `protoc` and no `buf` and no network. They are regenerated by
# `buf.gen.yaml` with pinned remote plugins and checked for drift in CI.

cmake_minimum_required(VERSION 3.20)
project(loams-cpp VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

find_package(Protobuf REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
# protobuf's JSON helpers (`google/protobuf/util/json_util.h`) return
# `absl::Status`, so anything that parses a Connect error body links Abseil — and
# Abseil's libraries are only on the link line through **pkg-config**, not through
# `protobuf::libprotobuf`, which names just `libprotobuf`. Linking the imported
# target alone therefore fails at the last step with "DSO missing from command
# line" for `libabsl_status.so`, which is a confusing way to learn that.
#
# So the link line comes from pkg-config, which is where protobuf's own `.pc` lists
# the whole set, and `find_package(Protobuf)` is kept for the include directories
# and the version check. `DEPENDENCIES.md` records why Abseil appears in the
# dependency list at all.
pkg_check_modules(LOAMS_PROTOBUF REQUIRED IMPORTED_TARGET protobuf)

# Warnings are errors. A generated stub is excluded from them: protoc's output is
# not this SDK's code to keep tidy, and a `-Werror` on a protobuf upgrade would
# break the build for no reason a reader could act on.
add_library(loams_warnings INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(loams_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()

# The committed stubs. `gen/` holds protoc's output verbatim for every proto
# package an SDK package needs; the error package is the one every other package
# needs and it is listed first so the include path is obvious.
file(GLOB LOAMS_GEN_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gen/loams/*/v1/*.pb.cc")
file(GLOB LOAMS_GEN_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gen/loams/*/v1/*.pb.h")
if(NOT LOAMS_GEN_SOURCES)
  message(FATAL_ERROR
    "sdks/cpp/gen has no generated stubs. They are committed; run "
    "`buf generate --template sdks/cpp/buf.gen.yaml` and commit the result.")
endif()

add_library(loams_gen STATIC ${LOAMS_GEN_SOURCES} ${LOAMS_GEN_HEADERS})
target_include_directories(loams_gen PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/gen>
  $<INSTALL_INTERFACE:include/loams-gen>)
target_include_directories(loams_gen SYSTEM PUBLIC ${Protobuf_INCLUDE_DIRS})
target_link_libraries(loams_gen PUBLIC PkgConfig::LOAMS_PROTOBUF)
# protoc's output is generated code, not this SDK's code: warnings stay off it.
set_source_files_properties(${LOAMS_GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "-w")

# The runtime and the hand-written facade (design §44 §7.3's Q604 fallback).
file(GLOB LOAMS_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_library(loams STATIC ${LOAMS_SOURCES})
target_include_directories(loams
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
         $<INSTALL_INTERFACE:include>
  PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(loams
  PUBLIC loams_gen PkgConfig::LOAMS_PROTOBUF
  PRIVATE CURL::libcurl Threads::Threads)
target_compile_options(loams PRIVATE $<$<BOOL:${CMAKE_CXX_COMPILER_ID}>:-Wno-unused-parameter>)
target_link_libraries(loams PRIVATE loams_warnings)

# --- Tests ------------------------------------------------------------------------------------------
#
# Six ctest tests, registered under the exact names `sdks/conformance/required.mjs`
# names for C++ (`cpp_conformance_all_required_fixtures` and its five siblings),
# plus a seventh that guards the set. The names are the contract between two
# languages' suites, so they are spelled out here rather than derived: a derived
# name would change the moment the helper that derives it changed.
#
# The **ctest name** and the **executable name** differ on purpose. ctest needs
# `cpp_conformance_all_required_fixtures`; the executable is `conformance_test`,
# because a target named `cpp_*` would put a C++ prefix on something that is not
# part of the SDK. `required.mjs` only ever sees the ctest name, which is what
# `run-test.sh` hands to `ctest -R`.
enable_testing()

# The shared test support: assertions, the corpus reader, the three-tier fixture
# endpoint and the results report. A library rather than a source file each test
# compiles, because it is one implementation of the replay and the JSON reading,
# and six copies of it would be six things to keep in step.
add_library(loams_test_support STATIC tests/support.cpp tests/required.cpp)
target_link_libraries(loams_test_support PUBLIC loams)
target_include_directories(loams_test_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(loams_test_support PRIVATE loams_warnings)

# The six, as `<canonical ctest name>|<executable stem>`.
set(LOAMS_TESTS
  cpp_conformance_all_required_fixtures|conformance_test
  cpp_error_reason_mapping|error_reason_test
  cpp_pagination_iterator|pagination_test
  cpp_retry_reuses_idempotency_key|retry_idempotency_test
  cpp_stream_resume_with_cursor|stream_resume_test
  cpp_token_source_refresh|token_source_test)

foreach(entry IN LISTS LOAMS_TESTS)
  string(REPLACE "|" ";" parts "${entry}")
  list(GET parts 0 ctest_name)
  list(GET parts 1 target)
  add_executable(${target} tests/${target}.cpp)
  target_link_libraries(${target} PRIVATE loams_test_support loams_warnings)
  add_test(NAME ${ctest_name} COMMAND ${target})
  # The suite resolves the corpus and the shared fixture server relative to the
  # source tree, and ctest runs in the build directory, so both are passed in
  # rather than searched for. That also makes `run-test.sh`, which runs `ctest`
  # from `sdks/cpp`, work regardless of where the build directory is.
  set_tests_properties(${ctest_name} PROPERTIES
    ENVIRONMENT
      "LOAMS_FIXTURES_DIR=${CMAKE_CURRENT_SOURCE_DIR}/../fixtures;LOAMS_CONFORMANCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/../conformance")
endforeach()

# The guard, registered under its own name because `required.mjs` names six and a
# seventh under one of their names would make this language's set differ from every
# other language's. It asserts that all six above are registered.
add_executable(conformance_test_names tests/test_names.cpp)
target_link_libraries(conformance_test_names PRIVATE loams loams_warnings)
add_test(NAME cpp_conformance_test_names COMMAND conformance_test_names)

# The quickstart is compiled, so a change that breaks it breaks the build rather
# than sitting in the tree broken until someone runs it by hand.
add_executable(quickstart examples/quickstart/main.cpp)
target_link_libraries(quickstart PRIVATE loams)

install(TARGETS loams loams_gen ARCHIVE DESTINATION lib)
install(DIRECTORY include/loams DESTINATION include)
install(DIRECTORY gen/loams DESTINATION include/loams-gen)
